## `ttlru_cache.py`

```python
from __future__ import annotations

import time
from collections import OrderedDict
from threading import RLock
from typing import Callable, Generic, TypeVar

K = TypeVar("K")
V = TypeVar("V")

_MISSING = object()


class TTLRUCache(Generic[K, V]):
    """Thread-safe LRU cache with per-entry TTL expiration.

    - Expired entries are removed lazily (on access); no background threads.
    - All operations are O(1).
    - `get` returns None for missing or expired keys.
    """

    def __init__(
        self,
        maxsize: int,
        ttl_seconds: float,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        if maxsize <= 0:
            raise ValueError("maxsize must be a positive integer")
        if ttl_seconds <= 0:
            raise ValueError("ttl_seconds must be positive")
        self._maxsize = maxsize
        self._ttl = ttl_seconds
        self._clock = clock
        # key -> (value, expires_at); OrderedDict keeps LRU order
        self._data: OrderedDict[K, tuple[V, float]] = OrderedDict()
        self._lock = RLock()

    def get(self, key: K) -> V | None:
        with self._lock:
            entry = self._data.get(key, _MISSING)
            if entry is _MISSING:
                return None
            value, expires_at = entry  # type: ignore[misc]
            if expires_at <= self._clock():
                del self._data[key]  # lazy expiration
                return None
            self._data.move_to_end(key)  # mark as most recently used
            return value

    def set(self, key: K, value: V) -> None:
        with self._lock:
            if key in self._data:
                # refresh position, then overwrite below
                self._data.move_to_end(key)
            self._data[key] = (value, self._clock() + self._ttl)
            while len(self._data) > self._maxsize:
                self._data.popitem(last=False)  # evict LRU

    def delete(self, key: K) -> bool:
        with self._lock:
            return self._data.pop(key, _MISSING) is not _MISSING

    def clear(self) -> None:
        with self._lock:
            self._data.clear()

    def __len__(self) -> int:
        # Note: expired-but-unaccessed entries count until touched.
        # We purge them here for an accurate logical size.
        with self._lock:
            self._purge_expired()
            return len(self._data)

    def __contains__(self, key: K) -> bool:
        return self.get(key) is not None

    def _purge_expired(self) -> None:
        now = self._clock()
        expired = [k for k, (_, exp) in self._data.items() if exp <= now]
        for k in expired:
            del self._data[k]
```

A few deliberate decisions worth flagging:

- **`__len__` purges expired entries** so the reported size is logically correct; without it, `len` would count zombie entries. Slightly more than O(1), but arguably the right semantic — happy to change it if you'd rather keep `len` cheap and document zombies.
- **`set` on an existing key refreshes both recency and TTL.** Alternative: don't refresh TTL on overwrite. Refreshing seems least surprising.
- **`RLock`** so `__contains__` can compose with `get` safely.
- **`time.monotonic`** by default — immune to wall-clock jumps.
- **`get` returning `None`** means "absent", so storing `None` as a value is indistinguishable from a miss. That's per your interface; if it matters, we'd add a sentinel-based `get(key, default)`.

## Bugs / correctness issues

**1. Eviction ignores expired entries (`set`).** When full, you evict the LRU entry even if the cache contains expired zombies. Pathological case:

```python
c = TTLRUCache(maxsize=2, ttl_seconds=10, clock=clock)
c.set("a", 1); c.set("b", 2)
clock.advance(20)          # both expired
c.set("c", 3)              # evicts "a" (fine), len==2... 
c.set("d", 4)              # evicts "b" — but "a" was already a zombie;
                           # we evicted 2 dead entries one at a time, OK here,
                           # but with maxsize pressure + mixed live/dead,
                           # you can evict a LIVE entry while zombies sit there
```

The real failure: `set("a"); set("b")`, advance past TTL of `a` only (stagger the sets), then `set("c")` — if `a` happens not to be LRU you evict a live key while `a` is garbage. Fix: purge expired before evicting, or evict expired-first:

```python
def set(self, key: K, value: V) -> None:
    with self._lock:
        if key in self._data:
            del self._data[key]
        self._data[key] = (value, self._clock() + self._ttl)
        if len(self._data) > self._maxsize:
            self._purge_expired()          # cheap win first
        while len(self._data) > self._maxsize:
            self._data.popitem(last=False)
```

(Also note `move_to_end` + reassign is redundant; `del` + assign is clearer and identical in effect.)

**2. `__contains__` mutates recency.** `key in cache` shouldn't count as a "use" — it refreshes LRU position as a side effect, which will surprise anyone using it for existence checks before a conditional `set`. Write it without `get`:

## `ttlru_cache.py` (v2)

```python
from __future__ import annotations

import time
from collections import OrderedDict
from dataclasses import dataclass
from threading import RLock
from typing import Callable, Generic, TypeVar, overload

K = TypeVar("K")
V = TypeVar("V")
D = TypeVar("D")

_MISSING = object()


@dataclass(frozen=True)
class CacheStats:
    """Point-in-time snapshot of cache counters."""
    hits: int
    misses: int
    evictions: int        # LRU evictions due to maxsize pressure
    expirations: int      # entries removed due to TTL expiry
    size: int             # live entries at snapshot time


class TTLRUCache(Generic[K, V]):
    """Thread-safe LRU cache with per-entry TTL expiration.

    - Expired entries are removed lazily (on access or eviction pressure);
      no background threads.
    - get/set/delete are O(1) amortized. len() is O(n) worst case because
      it purges expired entries to report an accurate logical size.
    - `get` returns None for missing or expired keys; pass a sentinel as
      `default` if you need to distinguish a stored None from a miss.
    - Expired entries are reclaimed before any live entry is evicted.
    """

    def __init__(
        self,
        maxsize: int,
        ttl_seconds: float,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        if maxsize <= 0:
            raise ValueError("maxsize must be a positive integer")
        if ttl_seconds <= 0:
            raise ValueError("ttl_seconds must be positive")
        self._maxsize = maxsize
        self._ttl = ttl_seconds
        self._clock = clock
        self._data: OrderedDict[K, tuple[V, float]] = OrderedDict()
        self._lock = RLock()
        self._hits = 0
        self._misses = 0
        self._evictions = 0
        self._expirations = 0

    # --- internal helpers (must be called with self._lock held) ---

    def _is_expired(self, expires_at: float) -> bool:
        return expires_at <= self._clock()

    def _drop_expired_entry(self, key: K) -> None:
        del self._data[key]
        self._expirations += 1

    def _purge_expired(self) -> None:
        now = self._clock()
        expired = [k for k, (_, exp) in self._data.items() if exp <= now]
        for k in expired:
            del self._data[k]
        self._expirations += len(expired)

    # --- public API ---

    @overload
    def get(self, key: K) -> V | None: ...
    @overload
    def get(self, key: K, default: D) -> V | D: ...

    def get(self, key: K, default: object = None) -> object:
        with self._lock:
            entry = self._data.get(key, _MISSING)
            if entry is _MISSING:
                self._misses += 1
                return default
            value, expires_at = entry  # type: ignore[misc]
            if self._is_expired(expires_at):
                self._drop_expired_entry(key)
                self._misses += 1
                return default
            self._data.move_to_end(key)
            self._hits += 1
            return value

    def set(self, key: K, value: V) -> None:
        with self._lock:
            # del + assign: refreshes both recency and TTL
            if key in self._data:
                del self._data[key]
            self._data[key] = (value, self._clock() + self._ttl)
            if len(self._data) > self._maxsize:
                # reclaim zombies before evicting anything live
                self._purge_expired()
            while len(self._data) > self._maxsize:
                self._data.popitem(last=False)
                self._evictions += 1

    def delete(self, key: K) -> bool:
        """Remove key. Returns True only if a *live* entry was removed."""
        with self._lock:
            entry = self._data.get(key, _MISSING)
            if entry is _MISSING:
                return False
            if self._is_expired(entry[1]):  # type: ignore[index]
                self._drop_expired_entry(key)
                return False
            del self._data[key]
            return True

    def clear(self) -> None:
        with self._lock:
            self._data.clear()

    def __len__(self) -> int:
        with self._lock:
            self._purge_expired()
            return len(self._data)

    def __contains__(self, key: K) -> bool:
        """Existence check; does NOT refresh LRU recency."""
        with self._lock:
            entry = self._data.get(key, _MISSING)
            if entry is _MISSING:
                return False
            if self._is_expired(entry[1]):  # type: ignore[index]
                self._drop_expired_entry(key)
                return False
            return True

    def stats(self) -> CacheStats:
        with self._lock:
            self._purge_expired()
            return CacheStats(
                hits=self._hits,
                misses=self._misses,
                evictions=self._evictions,
                expirations=self._expirations,
                size=len(self._data),
            )
```

## Property-based tests — `test_ttlru_cache_hypothesis.py`

The model: a plain `dict` of `key -> (value, expires_at)`, replayed against the same `FakeClock`. The trick is that the model must replicate *lazy* expiration semantics, which is easiest by making the model purge-on-read like the cache does — but critically, the model is written naively enough that an LRU-ordering bug in the real cache (like v1's) still shows up as a divergence in which keys survive.

```python
from hypothesis import given, settings
from hypothesis import strategies as st


class ModelCache:
    """Naive reference implementation. Correct by inspection;
    independently implements lazy expiry + LRU so bugs in the real
    cache's ordering/expiry logic surface as divergences."""

    def __init__(self, maxsize: int, ttl: float, clock) -> None:
        self.maxsize = maxsize
        self.ttl = ttl
        self.clock = clock
        self.data: dict[str, tuple[int, float]] = {}

    def _live_keys(self) -> list[str]:
        now = self.clock()
        return [k for k, (_, exp) in self.data.items() if exp > now]

    def get(self, key: str):
        entry = self.data.get(key)
        if entry is None:
            return None
        value, exp = entry
        if exp <= self.clock():
            del self.data[key]
            return None
        self.data[key] = self.data.pop(key)  # move to end
        return value

    def set(self, key: str, value: int) -> None:
        if key in self.data:
            del self.data[key]
        self.data[key] = (value, self.clock() + self.ttl)
        # reclaim expired first, then evict LRU
        for k in [k for k, (_, e) in self.data.items() if e <= self.clock()]:
            del self.data[k]
        while len(self.data) > self.maxsize:
            del self.data[next(iter(self.data))]

    def delete(self, key: str) -> bool:
        entry = self.data.get(key)
        if entry is None:
            return False
        del self.data[key]
        return entry[1] > self.clock()

    def live_contents(self) -> dict[str, int]:
        return {k: v for k, (v, e) in self.data.items() if e > self.clock()}
```
